cmake_minimum_required(VERSION 3.22)
project(DurowMenuOnly LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB MENU_SOURCES CONFIGURE_DEPENDS menu/*.cpp)
add_library(durow_menu STATIC ${MENU_SOURCES} assets.cpp font_data.cpp
 third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
 third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp)
target_include_directories(durow_menu PUBLIC . third_party/imgui PRIVATE third_party)
target_compile_definitions(durow_menu PUBLIC KAMSHAKO_PORTABLE_MENU IMGUI_DEFINE_MATH_OPERATORS NOMINMAX PRIVATE MENU_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(MSVC)
 target_compile_options(durow_menu PRIVATE /utf-8 /W3)
 target_compile_definitions(durow_menu PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(menu_smoke smoke.cpp)
target_link_libraries(menu_smoke PRIVATE durow_menu)
enable_testing()
add_test(NAME menu_ui_only COMMAND menu_smoke)
set_tests_properties(menu_ui_only PROPERTIES ENVIRONMENT "NL_DATA_DIR=${CMAKE_CURRENT_BINARY_DIR}/test-data")
